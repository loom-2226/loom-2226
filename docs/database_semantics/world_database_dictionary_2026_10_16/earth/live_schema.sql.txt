   FROM (loom_control.snapshot s
     JOIN loom_control.model_context_record m ON (((m.snapshot_id = s.snapshot_id) AND (m.context_key = 'model_designation'::text))))
  WHERE ((s.state = 'VALIDATED'::text) AND ((m.context_value #>> '{}'::text[]) = 'EARTH_LEAN_BIOSYNTHETIC_COUPLED_SUCCESSOR_v0_1_2026_09_24'::text))

CREATE VIEW loom_narrator.current_earth_year AS
 WITH sid AS (
         SELECT current_earth_snapshot.snapshot_id
           FROM loom_narrator.current_earth_snapshot
        ), d AS (
         SELECT earth_demographic_year.year,
            sum(earth_demographic_year.biological_population) AS biological_population,
            sum(earth_demographic_year.age_under_20) AS age_under_20,
            sum(earth_demographic_year.age_20_64) AS age_20_64,
            sum(earth_demographic_year.age_65_plus) AS age_65_plus,
            sum(earth_demographic_year.age_80_plus) AS age_80_plus,
            sum(earth_demographic_year.age_100_plus) AS age_100_plus,
            sum(earth_demographic_year.age_120_plus) AS age_120_plus,
            sum(earth_demographic_year.age_150_plus) AS age_150_plus
           FROM (loom_earth.earth_demographic_year
             JOIN sid sid_1 USING (snapshot_id))
          GROUP BY earth_demographic_year.year
        ), e AS (
         SELECT earth_economic_year.year,
            sum(earth_economic_year.value_added) AS value_added,
            sum(earth_economic_year.gross_output) AS gross_output,
            sum(earth_economic_year.investment) AS investment,
            sum(earth_economic_year.capital) AS capital
           FROM (loom_earth.earth_economic_year
             JOIN sid sid_1 USING (snapshot_id))
          GROUP BY earth_economic_year.year
        ), l AS (
         SELECT earth_labor_composition_year.year,
            sum(earth_labor_composition_year.synthetic_population) AS synthetic_population,
            sum(earth_labor_composition_year.synthetic_effective_labor) AS synthetic_effective_labor,
            sum(earth_labor_composition_year.biological_effective_labor) AS biological_effective_labor,
            sum(earth_labor_composition_year.machine_task_capacity) AS machine_task_capacity,
            sum(earth_labor_composition_year.total_effective_labor) AS total_effective_labor
           FROM (loom_earth.earth_labor_composition_year
             JOIN sid sid_1 USING (snapshot_id))
          GROUP BY earth_labor_composition_year.year
        )
 SELECT sid.snapshot_id,
    d.year,
    d.biological_population,
    l.synthetic_population,
    (d.biological_population + l.synthetic_population) AS recognized_person_population,
    d.age_under_20,
    d.age_20_64,
    d.age_65_plus,
    d.age_80_plus,
    d.age_100_plus,
    d.age_120_plus,
    d.age_150_plus,
    l.biological_effective_labor,
    l.synthetic_effective_labor,
    l.machine_task_capacity,
    l.total_effective_labor,
    e.value_added,
    e.gross_output,
    e.investment,
    e.capital,
        CASE
            WHEN (d.year <= 2100) THEN 'WPP_2024_MEDIUM'::text
            ELSE 'MED_CENTRAL'::text
        END AS demographic_status,
        CASE
            WHEN (d.year <= 2100) THEN 'V4_PREDECESSOR'::text
            ELSE 'MED_CENTRAL__SYNTH_CENTRAL'::text
        END AS economic_status
   FROM (((sid
     CROSS JOIN d)
     LEFT JOIN e USING (year))
