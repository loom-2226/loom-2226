      LEFT JOIN civ_country_operating_profile p ON p.country_subject_id=s.subject_id
      LEFT JOIN civ_country_numeric_state n ON n.subject_id=s.subject_id AND n.year=2226 AND n.scenario_id='BLIND_MODEL_CONTINUATION'
      WHERE s.subject_class IN ('COUNTRY_LINEAGE','DISPUTED_POLITY_LINEAGE');

CREATE VIEW v_graph_civstate_actor_exposure AS
SELECT
    e.actor_subject_id,
    a.display_name AS actor_name,
    CASE
      WHEN a.navigator_entity_id IS NOT NULL THEN 'ENTITY:' || a.navigator_entity_id
      WHEN a.navigator_noun_id IS NOT NULL THEN 'NOUN:' || a.navigator_noun_id
      ELSE NULL
    END AS actor_node_key,
    e.geography_subject_id,
    g.display_name AS geography_name,
    CASE
      WHEN g.navigator_entity_id IS NOT NULL THEN 'ENTITY:' || g.navigator_entity_id
      WHEN g.navigator_noun_id IS NOT NULL THEN 'NOUN:' || g.navigator_noun_id
      ELSE NULL
    END AS geography_node_key,
    e.sector_id,
    e.year,
    e.employment_share,
    e.asset_share,
    e.revenue_share,
    e.control_weight,
    e.service_dependency_weight,
    e.derivation_id
FROM civ_actor_exposure e
JOIN civ_subject a ON a.subject_id=e.actor_subject_id
JOIN civ_subject g ON g.subject_id=e.geography_subject_id;

CREATE VIEW v_graph_civstate_influence_edges AS
SELECT
    i.subject_id,
    s.display_name AS subject_name,
    CASE
      WHEN s.navigator_entity_id IS NOT NULL THEN 'ENTITY:' || s.navigator_entity_id
      WHEN s.navigator_noun_id IS NOT NULL THEN 'NOUN:' || s.navigator_noun_id
      ELSE NULL
    END AS subject_node_key,
    i.actor_subject_id,
    a.display_name AS actor_name,
    CASE
      WHEN a.navigator_entity_id IS NOT NULL THEN 'ENTITY:' || a.navigator_entity_id
      WHEN a.navigator_noun_id IS NOT NULL THEN 'NOUN:' || a.navigator_noun_id
      ELSE NULL
    END AS actor_node_key,
    i.year,
    i.influence_domain,
    i.influence_weight,
    i.control_class,
    i.basis,
    i.derivation_id
FROM civ_influence_edge i
JOIN civ_subject s ON s.subject_id=i.subject_id
JOIN civ_subject a ON a.subject_id=i.actor_subject_id;

CREATE VIEW v_graph_link_catalog AS
SELECT
    e.link_key,
    e.from_node_key,
    fn.canonical_name AS from_name,
    fn.node_domain AS from_domain,
    e.to_node_key,
    tn.canonical_name AS to_name,
    tn.node_domain AS to_domain,
    e.relationship_type,
    e.relationship_domain,
    e.directionality,
    e.is_derived,
    e.context_entity_id,
    e.support_count,
    e.source_record_type,
    e.source_record_id,
    e.provenance,
    e.derivation_basis,
    x.display_label_override,
    x.summary_override,
    x.confidence_override,
    x.salience,
    x.metadata_json
FROM nav_graph_edges_cache e
JOIN nav_graph_nodes_cache fn ON fn.node_key=e.from_node_key
JOIN nav_graph_nodes_cache tn ON tn.node_key=e.to_node_key
LEFT JOIN graph_link_enrichment x ON x.link_key=e.link_key;

CREATE VIEW v_graph_nodes AS
SELECT * FROM nav_graph_nodes_cache;

CREATE VIEW v_graph_traversal AS
SELECT
    link_key,
    from_node_key,
    to_node_key,
    relationship_type,
    relationship_domain,
    is_derived,
    context_entity_id,
    support_count,
    0 AS reversed
FROM nav_graph_edges_cache
UNION ALL
SELECT
    link_key || ':REV',
    to_node_key,
    from_node_key,
    relationship_type,
    relationship_domain,
    is_derived,
    context_entity_id,
    support_count,
    1
FROM nav_graph_edges_cache;
